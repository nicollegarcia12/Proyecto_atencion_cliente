from fastapi import FastAPI, Form, Request
from fastapi.responses import HTMLResponse
from sqlalchemy import create_engine, Column, Integer, String, Text, DateTime
from sqlalchemy.orm import declarative_base, sessionmaker
from datetime import datetime
import uvicorn

# 1. CONFIGURACIÓN DE LA BASE DE DATOS (SQLite - se crea sola)
engine = create_engine("sqlite:///./portal.db")
SessionLocal = sessionmaker(bind=engine)
Base = declarative_base()

# 2. MODELO DE DATOS (Cómo se guarda la información)
class Ticket(Base):
    __tablename__ = "tickets"
    id = Column(Integer, primary_key=True, index=True)
    nombre_cliente = Column(String(100), nullable=False)
    problema = Column(Text, nullable=False)
    solucion_ia = Column(Text, nullable=True)
    fecha = Column(DateTime, default=datetime.utcnow)

# Creamos las tablas en la base de datos
Base.metadata.create_all(bind=engine)

# 4. CREACIÓN DEL PORTAL WEB (FastAPI)
app = FastAPI()

# Ruta para ver el Portal del Cliente (Interfaz web sencilla)

    html = """

            <p>Describa su problema. Se le dará seguimiento lo antes posible.</p>

            </form>

    </html>
    """

# Ruta para RECIBIR la información, guardarla en BD y activar la IA
@app.post("/enviar")
async def recibir_solicitud(nombre: str = Form(...), problema: str = Form(...)):
    # Nos conectamos a la base de datos
    db = SessionLocal()
    
    # La IA analiza el problema
    solucion = generar_solucion_ia(problema)
    
    # Guardamos todo en la Base de Datos
    nuevo_ticket = Ticket(
        nombre_cliente=nombre,
        problema=problema,
        solucion_ia=solucion
    )
    db.add(nuevo_ticket)
    db.commit()
    ticket_id = nuevo_ticket.id
    db.close()
    
    return {
        "mensaje": "Solicitud recibida exitosamente. Atenderemos tu caso pronto.",
        "ticket_id": ticket_id,
        "estado": "En Proceso"
    }

# Ruta para la EMPRESA (Ver solicitudes e informes)
@app.get("/admin/reportes")
def reporte_administracion():
    db = SessionLocal()
    tickets = db.query(Ticket).all()
    db.close()
    
    resultado = []
    for t in tickets:
        resultado.append({
            "ID": t.id,
            "Cliente": t.nombre_cliente,
            "Problema": t.problema,
            "Solución Sugerida por IA": t.solucion_ia,
            "Fecha": t.fecha
        })
    return {"total_tickets": len(resultado), "tickets": resultado}

# 5. ENCENDER EL SERVIDOR
if __name__ == "__main__":
    uvicorn.run(app, host="127.0.0.1", port=8000)
